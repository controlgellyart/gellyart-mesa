const scriptURL = 'PEGAR_AQUI_TU_URL_DE_GOOGLE_APPS_SCRIPT'; function enviarAlerta(tipo) { const urlParams = new URLSearchParams(window.location.search); const negocio = urlParams.get('negocio') || 'Desconocido'; const mesa = urlParams.get('mesa') || 'Desconocida'; const datos = { negocio: negocio, mesa: mesa, accion: tipo // "Llamar al Mesero" o "Pedir Cuenta" }; // Envío hacia Google Apps Script fetch(scriptURL, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(datos) }) .then(() => { alert("¡Solicitud enviada con éxito! El personal está en camino."); }) .catch(error => { console.error('Error:', error); alert("Hubo un problema al enviar la solicitud."); }); }
