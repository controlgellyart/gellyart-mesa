function enviarAccion(accion) { const msgDiv = document.getElementById('mensaje'); msgDiv.style.color = "#c69214"; msgDiv.innerText = "Enviando solicitud..."; const urlParams = new URLSearchParams(window.location.search); const nombreNegocio = urlParams.get('negocio') || 'GENERAL'; const numeroMesa = urlParams.get('mesa') || '1'; const URL_APPS_SCRIPT = "https://script.google.com/macros/s/AKfycbwHZTPuHu8cVLeBsGHSFD8L7M-DrX1rG5rIQ2NcR03RQ3i-TCQBaQ-8bh4Q0luXLskBBQ/exec"; fetch(URL_APPS_SCRIPT, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ negocio: nombreNegocio, mesa: numeroMesa, accion: accion }) }) .then(() => { msgDiv.style.color = "#2b8a3e"; msgDiv.innerText = "¡Solicitud enviada con éxito!"; setTimeout(() => { msgDiv.innerText = ""; }, 4000); }) .catch(error => { msgDiv.style.color = "#c92a2a"; msgDiv.innerText = "Error al enviar. Intenta de nuevo."; console.error(error); }); }
